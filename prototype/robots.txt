# ⚠️ 앱은 한 장짜리(SPA)다. 크롤러가 볼 것은 사실상 / 하나뿐이고,
#    나머지는 자료 파일이라 색인돼도 의미가 없다.
#
# ⚠️⚠️ **robots.txt 는 상속이 없다.**
#    크롤러는 **자기 이름의 블록이 있으면 그것만 읽고 `*` 블록을 통째로 무시한다.**
#    그래서 봇별 블록을 만들 때는 그 안에 Disallow 를 **다시 적어야 한다.**
#    (예전에 "GPTBot 을 명시 허용"하려고 블록만 추가했다가 그 봇에게만
#     자료 디렉터리가 통째로 열린 적이 있다.)
#    아래 학습봇 블록은 전부 `Disallow: /` 라서 이 함정에 걸리지 않는다.
#
# ══ AI 크롤러 정책 (2026-08-03 변경) ═══════════════════════════════
#
#  ⚠️⚠️ **학습은 막고, 검색 인용은 연다.** 둘은 다른 일이다.
#
#    · **학습(training)** — 우리 글과 자료를 모델 가중치에 녹인다.
#      출처가 사라지고, 우리가 만든 자료(해변 방위·낚시 지점·등산로·평년 분포)가
#      누구 것인지 알 수 없게 된다. → **막는다.**
#
#    · **검색 인용(retrieval)** — 사람이 물었을 때 우리 페이지를 읽고 **출처와 함께**
#      답한다. 이건 우리가 하려는 일 그 자체다 —
#      "지금 바다 수온이 어떤가"에 earthus 가 인용되는 것. → **연다.**
#
#  ⚠️ 이건 **자발적 준수**다. 지키지 않는 크롤러는 이 파일로 못 막는다.
#     실제 강제는 이용약관(제11조)과 서버 차단이 맡는다.
#
#  ⚠️ 되돌리려면 아래 "학습 전용" 블록을 지우면 된다. 그게 전부다.

# ── 검색·인용 봇 — 허용 ────────────────────────────────────────────
# ⚠️ 아래 봇들은 **자기 블록이 없어야** `*` 규칙을 따른다.
#    이름을 적는 순간 그 블록만 읽으므로, 허용하려면 **아무것도 안 적는 것**이 맞다.
#      OAI-SearchBot · ChatGPT-User · PerplexityBot · Perplexity-User ·
#      Applebot(일반) · Googlebot · Bingbot · DuckDuckBot · Yeti(네이버) · Daum
#    → 이들은 아래 `*` 블록을 그대로 따른다.

User-agent: *
Allow: /$
Allow: /index.html
# 소개 페이지 — 앱은 화면을 전부 JS 로 그리는 한 장짜리라 크롤러가 읽을 글이 적다.
# 이 페이지가 "earthus 가 무엇인가"에 답하는 정적 본문이다. 화면 캡처도 함께 연다.
Allow: /intro.html
Allow: /llms.txt
Allow: /shots/
Allow: /legal/

# ⚠️⚠️ 관리자 화면. noindex 메타도 넣었지만 그건 **읽어 주는 크롤러에게만** 통한다.
#    여기서도 막는다. ⚠️ 이건 보안이 아니라 예의다 —
#    실제 보호는 Supabase RLS 가 한다(admin.html 안의 SQL 참고).
Disallow: /admin.html

# 관측 자료는 색인 대상이 아니다.
# 색인되면 검색 결과에 JSON 원문이 뜨고, 크롤러가 대역폭만 먹는다.
Disallow: /wind/
Disallow: /events/
Disallow: /ocean/
Disallow: /solar/
Disallow: /clouds/
Disallow: /celestrak/
Disallow: /climate/

# ⚠️⚠️ /data/ 는 **우리가 만든 자료**다 — 공공자료를 그대로 옮긴 것이 아니다.
#    해변 방위 210곳 · 낚시 지점 946곳 · 등산로 12,982개 · 평년 분포 97지점 ·
#    활공장 17곳. OSM·기상청 원자료에서 우리가 계산·정리·검증해 만든 것이라
#    데이터베이스로 보호받는다(이용약관 제11조). 색인·수집 대상이 아니다.
Disallow: /data/

# ⚠️ /js/ 와 /css/ 는 **막지 않는다** (예전엔 막았다 — 되돌렸다).
#    구글은 페이지를 실제로 렌더해 보고 색인하는데, 스크립트와 스타일을 막으면
#    렌더가 실패해 빈 페이지로 판단한다. 우리처럼 화면을 전부 JS 가 그리는
#    앱에서는 이게 곧 "색인하지 마세요"와 같은 말이 된다.
#    자료(JSON)와 코드는 다르다 — 코드는 열어 둬야 한다.
