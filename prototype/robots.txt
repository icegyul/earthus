# earthus — https://earthus.net
#
# ⚠️ 앱은 한 장짜리(SPA)다. 크롤러가 볼 것은 사실상 / 하나뿐이고,
#    나머지는 자료 파일이라 색인돼도 의미가 없다.
#
# ⚠️⚠️ **규칙을 하나의 User-agent: * 블록에 모아 둔다. 봇별 블록을 만들지 말 것.**
#    robots.txt 는 상속이 없다 — 크롤러는 **자기 이름의 블록이 있으면 그것만 읽고
#    `*` 블록을 통째로 무시한다.** 그래서 "GPTBot 을 명시적으로 허용하자"고
#    봇별 블록을 추가하면, 그 봇에게만 아래 Disallow 가 사라져 자료 디렉터리가
#    통째로 열린다. (실제로 그렇게 썼다가 되돌렸다.)
#    AI 크롤러 허용은 아무것도 안 적는 것이 곧 허용이다 — 아래 설명 참고.
#
# ── AI 검색 크롤러 정책: 의도적으로 **허용** ──────────────────────
#    GPTBot·OAI-SearchBot·ClaudeBot·PerplexityBot·Google-Extended·
#    Applebot-Extended·CCBot 등을 따로 막지 않는다. robots.txt 에서
#    "안 막는 것"이 곧 허용이므로, 이들은 아래 `*` 규칙을 그대로 따른다.
#
#    왜 허용하는가: 우리는 공공 관측자료를 출처와 함께 전달하는 서비스다.
#    사람이 AI 에게 "지금 바다 수온이 어떤가"를 물을 때 우리가 인용되는 것이
#    곧 우리가 하려는 일이다. (반대 사례: space.skyrocket.de 는 이들을 전부
#    차단해 두어 우리가 그 위성 자료를 쓸 수 없었다. 차단은 그렇게 작동한다.)
#
#    ⚠️ 이 방침을 뒤집고 싶을 때만 봇별 블록을 만들 것. 그때는 위 상속 함정을
#       기억해 Disallow 목록을 그 블록 안에 **다시 적어야** 한다.

User-agent: *
Allow: /$
Allow: /index.html
# 소개 페이지 — 앱은 화면을 전부 JS 로 그리는 한 장짜리라 크롤러가 읽을 글이 적다.
# 이 페이지가 "earthus 가 무엇인가"에 답하는 정적 본문이다. 화면 캡처도 함께 연다.
Allow: /intro.html
Allow: /llms.txt
Allow: /shots/

# 관측 자료는 색인 대상이 아니다.
# 색인되면 검색 결과에 JSON 원문이 뜨고, 크롤러가 대역폭만 먹는다.
Disallow: /wind/
Disallow: /events/
Disallow: /ocean/
Disallow: /solar/
Disallow: /clouds/
Disallow: /celestrak/
Disallow: /data/

# ⚠️ /js/ 와 /css/ 는 **막지 않는다** (예전엔 막았다 — 되돌렸다).
#    구글은 페이지를 실제로 렌더해 보고 색인하는데, 스크립트와 스타일을 막으면
#    렌더가 실패해 빈 페이지로 판단한다. 우리처럼 화면을 전부 JS 가 그리는
#    앱에서는 이게 곧 "색인하지 마세요"와 같은 말이 된다.
#    자료(JSON)와 코드는 다르다 — 코드는 열어 둬야 한다.

Sitemap: https://earthus.net/sitemap.xml
